Part 5: Displaying Python variables in HTML with Jinja2

In this section I'll cover using Jinja2 to display variables from our Python code, which in our case will be our blog posts, in our HTML code. This will be a super quick tutorial as we're only going to be adding a few lines to our app.py and index.html files.

Starting with app.py, we're just going to add a variable in our index route that grabs each item in our BlogPost table that we added in the last section. We're also going to add a route for deleting blog posts from the database so we can do it all from our website without needing to go back and forth between it and Adminer. So let's follow along with the box below and get to work on that ^-^.

#!/usr/bin/env python3

from flask import Flask, render_template, redirect, request

from flask_migrate import Migrate

from flask_sqlalchemy import SQLAlchemy

import psycopg

from datetime import datetime

app = Flask(__name__)

app.config[ 'SQLALCHEMY_DATABASE_URI'] = "postgresql://user:pass@db:5432/db_name"

app.config[ 'SQLALCHEMY_TRACK_MODIFICATIONS'] = False

app.config[ 'SECRET_KEY'] = "bunnies r the bestest animalz ever !!11!!!!!!"

db = SQLAlchemy(app)

migrate = Migrate(app, db)

class BlogPost(db.Model):

     __tablename__ = 'blog_posts'

     id = db.Column(db.Integer, primary_key=True)

     title = db.Column(db.String, unique=True, nullable=False)

     date_posted = db.Column(db.DateTime, default=datetime.now())

     def __init__(self, title, date_posted):

          self.title = title

          self.date_posted = date_posted

     def __repr__(self):

          return f'Title: {self.title} Date: {self.date_posted}'

@app.route('/')

def index():

# All we have to add to our app file is this query all statement, as well as passing the variable into our template.

     posts = BlogPost.query.all()

# Passing the variable into the template will let us use Jinja2 to create a for loop to display all our blog posts

     return render_template('index.html', posts=posts)

@app.route('/blog/<int:id>')

def get_post(id):

     post = db.session.get(BlogPost, id)

     return render_template(f'{post.id}.html')

@app.route('/create-blog-post', methods=['POST'])

def add_post():

     title = request.form.get("title")

     date_posted = datetime.now()

     if title != "":

          post = BlogPost(title=title, date_posted=date_posted)

          db.session.add(post)

          db.session.commit()

          return redirect(f'/blog/{post.id}')

     else:

          return redirect('/')

# This route is basically the same as the one we made to create posts, but it takes in the posts id, queries the database for that id, and then deletes it.

@app.route('/delete-blog-post/<int:id>')

def delete_post(id):

          post = db.session.get(BlogPost, id)

          db.session.delete(post)

          db.session.commit()

          return redirect('/')

if __name__ == '__main__':

     with app.app_context():

          db.create_all()

     app.run(host="0.0.0.0", port=5000)

<html>

     <body>

<!-- This is where we will put our Jinja2 for loop. In the loop we will use our posts variable from our flask code and add a <p> element for each of our blog posts. -->

               {% for post in posts %}

                    <p> Post #{{post.id}} | {{post.title}} | Posted on: {{post.date_posted.day}} /{{post.date_posted.month}} /{{post.date_posted.year}}

<!-- A simple button that will call the delete route we made above with the post's id -->

                    <a href="/delete-blog-post/{post.id}"<button> Delete </button></a>

               {% endfor %}

          </main>

     </body>

</html>
